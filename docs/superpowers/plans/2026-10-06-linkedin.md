# LinkedIn as a second destination (manual mode) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Each piece gets a LinkedIn text written with the Instagram caption; at gate 2 Coni edits it and chooses Instagram, LinkedIn or both; at the approved time each destination waits for her to upload it by hand and is marked separately.

**Architecture:** Two columns on `piezas` (`texto_linkedin`, `destinos`), a gate-2 formula column, and a `publicaciones` table with one row per piece and destination. The piece keeps its single state machine (`para_subir_a_mano` means "some destination pending"). Gate 2 gets formula 2, which hashes caption, LinkedIn text and destinations; formula 1 keeps validating old approvals. Spec: `docs/superpowers/specs/2026-10-06-conidolly-linkedin-design.md` (decisions D1 to D10 are referenced below).

**Tech Stack:** Next.js 16 App Router, React 19, TypeScript strict, Supabase (Postgres + RLS, local for `test:db`), Trigger.dev v4, Vitest + Testing Library, Playwright, Biome (safe mode), Calma suiza components (`components/ui/*`) and tokens (`app/globals.css`).

-- LinkedIn as a second destination, by hand (design 2026-10-06, D1, D3, D10).
alter table public.piezas
  add column texto_linkedin text,
  add column destinos text[] not null default '{instagram}',
  add column puerta2_formula smallint,
  add constraint destinos_validos check (
    cardinality(destinos) between 1 and 2
    and destinos <@ array['instagram', 'linkedin']
    and cardinality(destinos) = cardinality(array(select distinct unnest(destinos)))
  ),
  add constraint puerta2_formula_valida check (puerta2_formula is null or puerta2_formula in (1, 2)),
  add constraint texto_linkedin_largo check (texto_linkedin is null or char_length(texto_linkedin) <= 3000);

alter table public.ajustes
  add column destinos_por_defecto text[] not null default '{instagram}',
  add constraint destinos_por_defecto_validos check (
    cardinality(destinos_por_defecto) between 1 and 2
    and destinos_por_defecto <@ array['instagram', 'linkedin']
    and cardinality(destinos_por_defecto) = cardinality(array(select distinct unnest(destinos_por_defecto)))
  );

create table public.publicaciones (
  pieza_id uuid not null references public.piezas (id) on delete cascade,
  destino text not null check (destino in ('instagram', 'linkedin')),
  estado text not null check (estado in ('esperando', 'para_subir', 'subido_a_mano', 'publicado')),
  enlace text,
  marcado_en timestamptz,
  marcado_por uuid,
  primary key (pieza_id, destino)
);
alter table public.publicaciones enable row level security;
create policy publicaciones_lee_coni on public.publicaciones for select using (public.es_coni());
revoke insert, update, delete on public.publicaciones from authenticated, anon;
create trigger auditar_publicaciones after insert or update or delete on public.publicaciones
  for each row execute function public.auditar();

-- Existing pieces keep exactly today's behavior: Instagram only, formula 1.
insert into public.publicaciones (pieza_id, destino, estado, enlace)
select id, 'instagram',
  case estado
    when 'programado' then 'esperando'
    when 'para_subir_a_mano' then 'para_subir'
    when 'publicado_a_mano' then 'subido_a_mano'
    when 'publicado' then 'publicado'
  end,
  permalink
from public.piezas
where estado in ('programado', 'para_subir_a_mano', 'publicado_a_mano', 'publicado');

export const DESTINOS = ["instagram", "linkedin"] as const
export type Destino = (typeof DESTINOS)[number]
export type EstadoDePublicacion = "esperando" | "para_subir" | "subido_a_mano" | "publicado"
export interface Publicacion {
  destino: Destino
  estado: EstadoDePublicacion
}

const NOMBRE: Record<Destino, string> = { instagram: "Instagram", linkedin: "LinkedIn" }

export function esDestino(valor: string): valor is Destino {
  return (DESTINOS as readonly string[]).includes(valor)
}

export function destinosValidos(valores: readonly string[]): valores is readonly Destino[] {
  return valores.length > 0 && valores.every(esDestino) && new Set(valores).size === valores.length
}

/** The stored order, also the order inside the gate-2 hash: Instagram first. */
export function canonicos(destinos: readonly Destino[]): Destino[] {
  return DESTINOS.filter((destino) => destinos.includes(destino))
}

export function nombreDeDestinos(destinos: readonly Destino[]): string {
  return canonicos(destinos).map((destino) => NOMBRE[destino]).join(" y ")
}

export function resumenDePublicaciones(publicaciones: readonly Publicacion[]) {
  const pendientes = canonicos(
    publicaciones.filter((p) => p.estado === "esperando" || p.estado === "para_subir").map((p) => p.destino),
  )
  return {
    pendientes,
    todosListos: pendientes.length === 0,
    algunoAMano: publicaciones.some((p) => p.estado === "subido_a_mano" || p.estado === "para_subir"),
  }
}

import { canonicos, type Destino } from "@/lib/dominio/destinos"

export type FormulaPuerta2 = 1 | 2

/**
 * Formula 1 hashed the caption alone. Formula 2 (2026-10-06) adds the LinkedIn text and the destinations, so nothing
 * of what Coni approved can change after gate 2. The LinkedIn text only counts when LinkedIn is chosen.
 */
export function hashTextoDePuerta2(
  formula: FormulaPuerta2,
  lado: { texto: string; textoLinkedin: string | null; destinos: readonly Destino[] },
): string {
  if (formula === 1) return hashTexto(lado.texto)
  const destinos = canonicos(lado.destinos)
  const linkedin = destinos.includes("linkedin") ? (lado.textoLinkedin ?? "") : ""
  return sha256(["puerta2/2", lado.texto, linkedin, destinos.join(",")])
}

/** A null stored formula on an approved piece is a gate-2 approval from before 2026-10-06: formula 1. */
export function formulaDePuerta2(guardada: number | null): FormulaPuerta2 {
  return guardada === 2 ? 2 : 1
}

export const MAXIMO_HASHTAGS_LINKEDIN = 3
const HASHTAG = /(^|\s)#[\p{L}\p{N}_]+/gu

/** The LinkedIn text is published as written, like the caption: same blocking rules, plus LinkedIn's own limits. */
export function revisarTextoLinkedin(entrada: {
  texto: string
  esAvatar: boolean
  requiereLineasAyuda: boolean
  ajeno: Ajeno
}): Hallazgo[] {
  const { texto, esAvatar, requiereLineasAyuda, ajeno } = entrada
  const hallazgos: Hallazgo[] = []
  if (esAvatar && !texto.includes(AVISO_AVATAR)) {
    hallazgos.push({ regla: "aviso_avatar", tipo: "bloquea", mensaje: `El texto para LinkedIn tiene que avisar que es un avatar digital: «${AVISO_AVATAR}»` })
  }
  if (requiereLineasAyuda && !["1455", "*4141", "133"].every((linea) => texto.includes(linea))) {
    hallazgos.push({ regla: "lineas_ayuda", tipo: "bloquea", mensaje: "Este tema pide las líneas de ayuda de Chile también en LinkedIn: 1455, *4141 y 133." })
  }
  if ([...texto].length > MAXIMO_TEXTO_LINKEDIN) {
    hallazgos.push({ regla: "largo_linkedin", tipo: "bloquea", mensaje: "LinkedIn acepta hasta 3.000 caracteres. Acórtalo un poco." })
  }
  const hashtags = texto.match(HASHTAG)?.length ?? 0
  if (hashtags > MAXIMO_HASHTAGS_LINKEDIN) {
    hallazgos.push({ regla: "hashtags_linkedin", tipo: "bloquea", mensaje: `En LinkedIn van como máximo ${MAXIMO_HASHTAGS_LINKEDIN} hashtags. Tiene ${hashtags}.` })
  }
  for (const frase of new Set(etiquetasDe(texto).filter((m) => m.conocida).map((m) => m.escrita))) {
    hallazgos.push({ regla: "etiqueta_en_texto", tipo: "bloquea", mensaje: "Las marcas de tono van solo en el guion. Bórrala del texto para LinkedIn.", frase })
  }
  for (const frase of new Set(tramosCopiados(texto, ajeno.citas))) {
    hallazgos.push({ regla: "copia_de_fuente", tipo: "advierte", mensaje: MENSAJE_COPIA, frase })
  }
  for (const frase of new Set(autoresNombrados(texto, ajeno.autores, ajeno.personas))) {
    hallazgos.push({ regla: "nombra_autor", tipo: "advierte", mensaje: MENSAJE_AUTOR, frase })
  }
  return hallazgos.map((hallazgo) => ({ ...hallazgo, campo: "linkedin" as const }))
}

create function public.editar_texto_linkedin_como_coni(p_pieza uuid, p_version integer, p_texto text, p_revision jsonb)
returns void language plpgsql security definer set search_path = '' as $$
declare
  fila public.piezas;
begin
  if not public.es_coni() then raise exception using errcode = '42501', message = 'solo_coni'; end if;
  select * into fila from public.piezas where id = p_pieza for update;
  if fila.version <> p_version then raise exception using errcode = '40001', message = 'version_desfasada'; end if;
  if fila.estado not in ('guion_por_aprobar', 'guion_aprobado', 'video_por_aprobar', 'programado') then
    raise exception using errcode = '42501', message = 'estado_no_editable';
  end if;
  if fila.estado = 'programado' then
    delete from public.publicaciones where pieza_id = p_pieza;
    update public.piezas set texto_linkedin = p_texto, revision = p_revision, estado = 'video_por_aprobar',
      puerta2_hash_texto = null, puerta2_video_id = null, puerta2_publicar_en = null, puerta2_en = null,
      puerta2_formula = null, intento = intento + 1, version = version + 1
    where id = p_pieza;
  else
    update public.piezas set texto_linkedin = p_texto, revision = p_revision, version = version + 1 where id = p_pieza;
  end if;
end $$;

create function public.marcar_subido_como_coni(p_pieza uuid, p_version integer, p_destino text, p_enlace text)
returns void language plpgsql security definer set search_path = '' as $$
declare
  fila public.piezas;
  quedan integer;
begin
  if not public.es_coni() then raise exception using errcode = '42501', message = 'solo_coni'; end if;
  select * into fila from public.piezas where id = p_pieza for update;
  if fila.version <> p_version then raise exception using errcode = '40001', message = 'version_desfasada'; end if;
  if fila.estado not in ('programado', 'para_subir_a_mano') then
    raise exception using errcode = '42501', message = 'estado_no_marcable';
  end if;
  update public.publicaciones set estado = 'subido_a_mano', enlace = p_enlace, marcado_en = now(), marcado_por = auth.uid()
  where pieza_id = p_pieza and destino = p_destino and estado in ('esperando', 'para_subir');
  if not found then raise exception using errcode = '42501', message = 'destino_no_pendiente'; end if;
  select count(*) into quedan from public.publicaciones
  where pieza_id = p_pieza and estado in ('esperando', 'para_subir');
  update public.piezas set
    estado = case when quedan = 0 then 'publicado_a_mano'::public.estado_pieza else estado end,
    permalink = case when p_destino = 'instagram' and p_enlace is not null then p_enlace else permalink end,
    version = version + 1
  where id = p_pieza;
end $$;
